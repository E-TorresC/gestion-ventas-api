-- Creación de la base de datos
CREATE DATABASE gestion_ventas_db;
GO

USE gestion_ventas_db;
GO

-- Tabla CLIENTE
CREATE TABLE cliente (
    id_cliente BIGINT IDENTITY(1,1) PRIMARY KEY,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    telefono VARCHAR(20),
    estado BIT DEFAULT 1,
    fecha_registro DATETIME DEFAULT GETDATE()
);
GO

-- Tabla CATEGORIA
CREATE TABLE categoria (
    id_categoria BIGINT IDENTITY(1,1) PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    estado BIT DEFAULT 1
);
GO

-- Tabla PRODUCTO
CREATE TABLE producto (
    id_producto BIGINT IDENTITY(1,1) PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    precio DECIMAL(10,2) NOT NULL CHECK (precio >= 0),
    stock INT NOT NULL DEFAULT 0 CHECK (stock >= 0),
    estado BIT DEFAULT 1,
    id_categoria BIGINT NOT NULL,
    CONSTRAINT fk_producto_categoria FOREIGN KEY (id_categoria) 
        REFERENCES categoria(id_categoria)
);
GO

-- Tabla PEDIDO
CREATE TABLE pedido (
    id_pedido BIGINT IDENTITY(1,1) PRIMARY KEY,
    id_cliente BIGINT NOT NULL,
    fecha_pedido DATETIME DEFAULT GETDATE(),
    total DECIMAL(12,2) NOT NULL DEFAULT 0,
    estado VARCHAR(20) NOT NULL DEFAULT 'PENDIENTE',
    CONSTRAINT fk_pedido_cliente FOREIGN KEY (id_cliente) 
        REFERENCES cliente(id_cliente),
    CONSTRAINT chk_pedido_estado CHECK (estado IN ('PENDIENTE', 'CONFIRMADO', 'ENVIADO', 'ENTREGADO', 'ANULADO'))
);
GO

-- Tabla DETALLE_PEDIDO
CREATE TABLE detalle_pedido (
    id_detalle_pedido BIGINT IDENTITY(1,1) PRIMARY KEY,
    id_pedido BIGINT NOT NULL,
    id_producto BIGINT NOT NULL,
    cantidad INT NOT NULL CHECK (cantidad > 0),
    precio_unitario DECIMAL(10,2) NOT NULL CHECK (precio_unitario >= 0),
    subtotal DECIMAL(12,2) NOT NULL CHECK (subtotal >= 0),
    CONSTRAINT fk_detalle_pedido FOREIGN KEY (id_pedido) 
        REFERENCES pedido(id_pedido),
    CONSTRAINT fk_detalle_producto FOREIGN KEY (id_producto) 
        REFERENCES producto(id_producto)
);
GO

-- Índices para mejorar rendimiento
CREATE INDEX idx_producto_categoria ON producto(id_categoria);
CREATE INDEX idx_pedido_cliente ON pedido(id_cliente);
CREATE INDEX idx_pedido_fecha ON pedido(fecha_pedido);
CREATE INDEX idx_detalle_pedido ON detalle_pedido(id_pedido);
CREATE INDEX idx_detalle_producto ON detalle_pedido(id_producto);
GO
